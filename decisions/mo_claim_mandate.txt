# MandateOverhaul
#
# The mandate is no longer taken by war. cb_take_mandate is is_triggered_only
# in common/cb_types/00_cb_types.txt, so nobody can declare with it; the
# mandate moves through this decision instead.
#
# The problem being solved: vanilla lets any neighbour of the Emperor claim the
# mandate by war, with no test of whether they could actually rule China. When
# Ming collapses, a two-province successor takes it, gets 25 years of
# new_mandate_holder, fails to unify in time, and then bleeds mandate from
# having no tributaries, war devastation, disaster events aimed at a unified
# Ming, and not holding the required cities. It loses the mandate, the next
# minor takes it, and the loop repeats for the rest of the campaign.
#
# So the bar is possession, not conquest: hold the three historical capitals as
# cores and 400 total development, and you may proclaim. By then you already
# rule China, which is the point - the mandate should describe a fact rather
# than create one.
#
# Beijing, Nanjing and Xi'an come from Paradox's own deprecated empire_of_china
# decision in decisions/EmpireofChina.txt, which used those three plus Luoyang.
# Luoyang is omitted: four capitals across three regions made the bar so high
# that nothing short of total unification cleared it, and MandateOfHeavenTweaks
# already uses the Beijing / Nanjing / Xi'an set for the mandate triggered
# modifier, so this keeps one definition of "holds China" across both mods.
#
# No separate check that the old Emperor has lost the capitals is needed: the
# claimant must own all three as cores, so by definition nobody else holds them.
#
# The handoff itself needs no scripting. set_emperor_of_china fires vanilla's
# on_mandate_of_heaven_gained, which converts nomad tech and units to Chinese,
# turns a tribal or non-monarchy government into a monarchy, adds the
# celestial_empire reform directly (bypassing its own culture-group potential,
# which matters because manchu is evenks, not east_asian), sets mandate and
# meritocracy to 60, grants new_mandate_holder for 25 years, cores Beijing,
# Nanjing and Kaifeng, claims all of china_superregion, clears factions and
# removes lost_mandate_of_heaven. The old holder gets on_mandate_of_heaven_lost.
#
# The global flag clear is what makes a destroyed mandate recoverable - vanilla
# sets empire_of_china_dismantled in on_annexed when the Emperor is annexed, and
# the celestial_empire reform refuses to apply while it is set.

country_decisions = {

	mo_claim_mandate_of_heaven = {
		major = yes
		potential = {
			has_dlc = "Mandate of Heaven"
			normal_or_historical_nations = yes
			is_emperor_of_china = no
			is_colonial_nation = no
			is_revolutionary = no
			NOT = { has_reform = shogunate }
			OR = {
				religion_group = eastern
				religion_group = pagan
			}
		}
		provinces_to_highlight = {
			OR = {
				province_id = 1816
				province_id = 1821
				province_id = 700
			}
			NOT = { country_or_subject_holds = ROOT }
		}
		allow = {
			is_free_or_tributary_trigger = yes
			is_at_war = no
			total_development = 400
			owns_core_province = 1816		# Beijing
			owns_core_province = 1821		# Nanjing
			owns_core_province = 700		# Xi'an
		}
		effect = {
			set_emperor_of_china = ROOT
			if = {
				limit = { has_global_flag = empire_of_china_dismantled }
				clr_global_flag = empire_of_china_dismantled
			}
			add_adm_power = -100
			add_dip_power = -100
			add_mil_power = -100
		}
		ai_will_do = {
			factor = 1
		}
		ai_importance = 400
	}
}
