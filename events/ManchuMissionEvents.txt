# Koxinga Flees to Taiwan
country_event = {
	id = mch_missions.5
	title = "mch_missions.5.t"
	desc = "mch_missions.5.d"
	picture = NAVAL_MILITARY_eventPicture
	
	fire_only_once = yes
	major = yes
	
	trigger = {
		tag = MNG
		NOT = {
			total_development = 600
		}
		#@ Was: any taiwan_area province is_empty. Dropped. Koxinga's fleet did not
		#@ need Taiwan to be vacant - it took the island off the Dutch. Whoever
		#@ holds those provinces loses them and keeps the core, and may come back
		#@ for them if they think it worth the trouble.
	}
	mean_time_to_happen = {
		months = 120
	}
	immediate = {
		hidden_effect = {
			taiwan_area = {
				add_core = FRM				#@ no longer limited to empty provinces
				cede_province = FRM
				change_culture = chimin
			}
			#@ Coastal Fujian goes with him. Three provinces on Taiwan cannot
			#@ carry what the Zheng actually were - a maritime trading power that
			#@ raised a hundred thousand men and came within reach of Nanjing.
			#@ Fuzhou 669 and Quanzhou 1829 are the coastal pair; Jianning 2152
			#@ and Tingzhou 2153 are inland and stay behind, which is also what
			#@ leaves Min a capital at Jianning later.
			669 = {
				add_core = FRM
				cede_province = FRM
			}
			1829 = {
				add_core = FRM
				cede_province = FRM
			}
			FRM = {
				define_ruler = {
					name = "Koxinga"
					dynasty = "Zheng"
				}
				add_accepted_culture = polynesian
				add_harmonized_religion = animism
				change_government = monarchy
				add_government_reform = chinese_warlord
				add_navy_tradition = 80				#@ new
				add_mercantilism = 40				#@ new
				add_country_modifier = {			#@ new - see common/event_modifiers/
					name = mo_the_zheng_maritime_network
					duration = -1
				}
				#@ REMOVED deliberately - this was add_historical_friend = MNG, the same
				#@ back from MNG, and join_all_defensive_wars_of = MNG.
				#@
				#@ The Zheng did not consider themselves allies of the Ming; they
				#@ considered themselves TO BE the Ming, keeping the Yongli calendar and
				#@ Yongli titles long after the mainland was gone. Whichever Ming is
				#@ still on the map here is a different branch of the Zhu - a rival
				#@ claim to the same mandate, not a cause worth dying for. Binding
				#@ Formosa into its defensive wars made Dongning the client of a court
				#@ it would not have recognised, and in play it pulled the Zheng fleet
				#@ into mainland land wars it had no reason to join.
				#@
				#@ mo_dongning.1 in events/mo_dongning.txt is where this relationship
				#@ actually gets decided, and it already offers fighting on for the Ming
				#@ as one of four options rather than assuming it.
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.6
						cavalry = 0.3
						artillery = 0.1
					}
				}
				random_owned_province = {
					limit = {
						has_port = yes
					}
					build_to_forcelimit = {
						heavy_ship = 0.3
						light_ship = 0.3
						transport = 0.2
					}
				}
			}
		}
	}
	option = {
		name = "mch_missions.5.a"
		if = {
			limit = {
				is_emperor_of_china = yes
			}
			add_mandate = -10
		}
		else = {
			add_prestige = -10
		}
	}
	option = {
		name = "mch_missions.5.b"
		trigger = {
			ai = no
		}
		if = {
			limit = {
				is_emperor_of_china = yes
			}
			add_mandate = -10
		}
		else = {
			add_prestige = -10
		}
		switch_tag = FRM
	}
}

# Agreement with the Dalai Lama
country_event = {
	id = mch_missions.6
	title = "mch_missions.6.t"
	desc = "mch_missions.6.d"
	picture = RELIGION_eventPicture
	
	fire_only_once = yes
	
	is_triggered_only = yes
	
	option = { # Accept cooperation and conversion
		name = "mch_missions.6.a"
		if = {
			limit = {
				NOT = {
					religion = vajrayana
				}
			}
			change_religion = vajrayana
		}
		add_ruler_modifier = {
			name = "genghis_khan_reborn"
			duration = -1
		}
		add_country_modifier = {
			name = "tibetan_missionaries"
			duration = 18250
		}
	}
	option = { # Rule Tibet by force
		name = "mch_missions.6.b"
		custom_tooltip = mongol_tibetan_rulership_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					region = tibet_region
					religion = vajrayana
				}
				add_province_modifier = {
					name = "mongol_rule_tibet"
					duration = 7300
				}
			}
		}
	}
}

# Uniting the Mongols
country_event = {
	id = mch_missions.7
	title = "mch_missions.7.t"
	desc = "mch_missions.7.d"
	picture = MONGOL_BANNERS_eventPicture
	
	fire_only_once = yes
	
	is_triggered_only = yes
	
	option = { # unite the tribes
		name = "mch_missions.7.a"
		if = {
			limit = {
				NOT = {
					primary_culture = mongol
				}
			}
			change_primary_culture = mongol
			set_ruler_culture = mongol
			if = {
				limit = {
					has_heir = yes
				}
				set_heir_culture = mongol
			}
		}
		custom_tooltip = unite_mongol_culture_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					OR = {
						culture = chahar
						culture = khalkha
						culture = oirats
					}
				}
				change_culture = mongol
			}
		}
	}
	option = { # reorganize the tribes
		name = "mch_missions.7.b"
		if = {
			limit = {
				NOT = { accepted_culture = mongol }
				NOT = { primary_culture = mongol }
			}
			add_accepted_culture = mongol
		}
		if = {
			limit = {
				NOT = { accepted_culture = chahar }
				NOT = { primary_culture = chahar }
			}
			add_accepted_culture = chahar
		}
		if = {
			limit = {
				NOT = { accepted_culture = khalkha }
				NOT = { primary_culture = khalkha }
			}
			add_accepted_culture = khalkha
		}
		if = {
			limit = {
				NOT = { accepted_culture = oirats }
				NOT = { primary_culture = oirats }
			}
			add_accepted_culture = oirats
		}
		add_country_modifier = {
			name = "reorganized_mongols"
			duration = -1
		}
	}
}
